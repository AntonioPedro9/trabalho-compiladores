int passoNivelAcima = 2;
double offsetGlobal = 0.5;

fun int potenciaRecursiva(int base, int exp) {
    if (exp <= 0) {
        return 1;
    }
    return base * potenciaRecursiva(base, exp - 1);
}

fun double acumularComEscopoExterno(int n) {
    double total = offsetGlobal;
    int atual = n;
    while (atual > 0) {
        int ganho = potenciaRecursiva(passoNivelAcima, atual);
        total = total + ganho;
        atual = atual - 1;
    }
    return total;
}
