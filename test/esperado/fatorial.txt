Program
  FunctionDecl(int calcularFatorial)
    Param(int n)
    Block
      VarDecl(int resultado)
        IntLit(1)
      VarDecl(int contador)
        Id(n)
      While
        Bin(>)
          Id(contador)
          IntLit(1)
        Block
          Assign(resultado)
            Bin(*)
              Id(resultado)
              Id(contador)
          Assign(contador)
            Bin(-)
              Id(contador)
              IntLit(1)
      If
        Bin(&&)
          Bin(>=)
            Id(resultado)
            IntLit(1)
          Bin(!=)
            Id(resultado)
            IntLit(0)
        Block
          Return
            Id(resultado)
        Block
          Return
            IntLit(0)
